## Idempotent Laws
$$
\begin{align}
A \cup A  & = A \\
A \cap A  & = A
\end{align}
$$

## Commutative Laws
$$
\begin{align}
A \cup B & = B \cup A \\
A \cap B & = B \cap A
\end{align}
$$

## Associative Laws
$$
\begin{align}
(A \cup B) \cup C &= A \cup (B \cup C) \\
(A \cap B) \cap C &= A \cap (B \cap C)
\end{align}
$$

## Distributive Laws
$$
\begin{align}
A \cup (B \cap C) & = (A \cup B) \cap (A \cup C) \\
A \cap (B \cup C) & = (A \cap B) \cup (A \cap C)
\end{align}
$$

## Identity Laws
$$
\begin{align}
A \cup \emptyset & = A \\
A \cap U & = A
\end{align}
$$


## Domination Laws
$$
\begin{align}
A \cup U & = U \\
A \cap \emptyset & = \emptyset
\end{align}
$$

## Complement Laws
$$
\begin{align}
A \cup A^C & = U \\
A \cap A^C & = \emptyset \\
(A^C)^C & = A \\
U^C & = \emptyset \\
\emptyset^C & = U
\end{align}
$$

## De Morgan's Laws
$$
\begin{align}
(A \cup B)^C & = A^C \cap B^C \\
(A \cap B)^C & = A^C \cup B^C
\end{align}
$$

## Absorption Laws
$$
\begin{align}
A \cup (A \cap B) & = A \\
A \cap (A \cup B) & = A
\end{align}
$$

## Set Difference Laws
$$
\begin{align}
A - (B \cup C) & = (A - B) \cap (A - C) \\
A - (B \cap C) & = (A - B) \cup (A - C) \\
A - B & = A \cap B^C 
\end{align}
$$
